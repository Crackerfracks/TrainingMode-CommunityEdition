# Dream Land GIFs, Falcon. Side platforms about y 30.14, x 30.25 to 61.4
# (and mirrored); top platform about y 51.4, x -19 to 19. Positions are a
# first guess: tune them until the landing is the one named.
autorun

# NIL: falling with the double jump left, Jump Timing (Kind NIL) times the
# double jump so the fighter settles onto the right platform with no lag.
# A plain fall from high up lands too fast for a NIL. Tune x/y if the
# landing isn't "NIL".
script nil
preset load Defaults
set Cues/NIL Cues = 1
set Jump/Show = 1
set Jump/Kind = 0
air 10 80 right 0 0 0 0
sync nil
mark clip nil in
15
follow 150 s:0.6,0
30
mark clip nil out

# Spot cues compared on the same NIL: the path alone, the landing spot
# timer alone, Platform Glow alone.
script spot-path
preset load Defaults
set Cues/NIL Cues = 1
set Jump/Show = 1
set Jump/Kind = 0
set Paths/Landing Path = 1
set Timers/Fixed Strip = 0
set Timers/Landing Spot = 0
air 10 80 right 0 0 0 0
sync spot-path
mark clip spot-path in
15
follow 150 s:0.6,0
30
mark clip spot-path out

script spot-timer
